const courseData = {};
for (let i = 1; i <= 43; i++) {
    let catName = i <= 10 ? "Bozor Strukturasi" : (i <= 20 ? "Likvidlik" : (i <= 30 ? "Order Block & FVG" : "Kirish va Risk Menejment"));
    if (i === 41) catName = "Global Treyding";
    if (i >= 42) catName = "Sarvar Trading AI";
    
    courseData[i] = {
        cat: catName,
        title: i === 41 ? "41. Top Treyderlar Hozirgi Qarorlari" : (i === 42 ? "42. SARVAR TRADING: Global Yangiliklar" : (i === 43 ? "43. SMC AI Matrix: Yakuniy Kombinatsiya" : `${i}-Dars: SMC Professional Taktikasi`))
    };
}

let currentLessonId = 42; 
let currentTF = "H1";

const menu = document.getElementById('lessonsMenu');
let lastCat = "";
Object.keys(courseData).forEach(id => {
    const lesson = courseData[id];
    if (lesson.cat !== lastCat) {
        lastCat = lesson.cat;
        const head = document.createElement('div');
        head.className = 'module-header';
        head.innerText = lastCat;
        menu.appendChild(head);
    }
    const row = document.createElement('div');
    row.className = `lesson-row ${id == 42 ? 'active' : ''}`;
    row.id = `lesson-row-${id}`;
    row.onclick = () => {
        document.querySelectorAll('.lesson-row').forEach(el => el.classList.remove('active'));
        row.classList.add('active');
        currentLessonId = id;
        closePanels();
        updateUI();
    };
    row.innerHTML = `<span class="lesson-id">${id}</span><span>${lesson.title}</span>`;
    menu.appendChild(row);
});

function toggleSidebar() {
    document.getElementById('sidebarMenu').classList.toggle('open');
    document.getElementById('screenOverlay').classList.toggle('show');
}
function toggleAI() {
    document.getElementById('aiPanel').classList.toggle('open');
    document.getElementById('screenOverlay').classList.toggle('show');
}
function closePanels() {
    document.getElementById('sidebarMenu').classList.remove('open');
    document.getElementById('aiPanel').classList.remove('open');
    document.getElementById('screenOverlay').classList.remove('show');
}

function changeTF(tf) {
    currentTF = tf;
    document.querySelectorAll('.tf-btn').forEach(el => el.classList.remove('active'));
    document.getElementById(`tf-${tf}`).classList.add('active');
    loadTradingViewWidget();
}

function loadTradingViewWidget() {
    const symbol = document.getElementById('assetSelect').value;
    let tvInterval = '60';
    if (currentTF === 'M15') tvInterval = '15';
    if (currentTF === 'H1') tvInterval = '60';
    if (currentTF === 'D1') tvInterval = 'D';

    new TradingView.widget({
        "width": "100%",
        "height": "100%",
        "symbol": symbol,
        "interval": tvInterval,
        "timezone": "Etc/UTC",
        "theme": "dark",
        "style": "1",
        "locale": "en",
        "toolbar_bg": "#131722",
        "enable_publishing": false,
        "hide_side_toolbar": false,
        "allow_symbol_change": false,
        "container_id": "tradingview_live_chart",
        "studies": ["Volume@tv-basicstudies"]
    });
    updateUI();
}

function updateUI() {
    const asset = document.getElementById('assetSelect').value;
    const matrix = institutionalMatrix[asset][currentTF];

    document.getElementById('footerTitle').innerText = `${courseData[currentLessonId].title} [${asset.split(':')[1]} - ${currentTF}]`;
    document.getElementById('footerDesc').innerText = matrix.desc;
    document.getElementById('sourceBox').innerText = `🏛️ Banklar Siyosati: ${matrix.bank}`;
    document.getElementById('accuracyTag').innerText = matrix.signal === "WAIT" ? "Winrate: --%" : "Winrate: 96%";

    document.getElementById('mDirection').innerText = matrix.signal;
    document.getElementById('mEntry').innerText = matrix.entry;
    document.getElementById('mSL').innerText = matrix.sl;
    document.getElementById('mTP').innerText = matrix.tp;

    const dirEl = document.getElementById('mDirection');
    if(matrix.signal === "BUY") dirEl.style.color = "var(--green)";
    else if(matrix.signal === "SELL") dirEl.style.color = "var(--red)";
    else dirEl.style.color = "var(--orange)";

    const sigCon = document.getElementById('signalContainer');
    sigCon.innerHTML = '';
    const badge = document.createElement('div');
    badge.className = `signal-badge signal-${matrix.signal.toLowerCase()}`;
    badge.innerText = `TAVSIYA: ${matrix.signal}`;
    sigCon.appendChild(badge);
}
